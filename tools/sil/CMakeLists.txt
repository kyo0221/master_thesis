cmake_minimum_required(VERSION 3.8)
project(sil_host)
set(CMAKE_CXX_STANDARD 17)
find_package(rclcpp REQUIRED)
find_package(path_smoother REQUIRED)
find_package(local_planner REQUIRED)
find_package(speed_path_planner REQUIRED)
find_package(trajectory_follower REQUIRED)
find_package(traffic_signal_stop REQUIRED)
add_executable(sil_host sil_main.cpp)
target_link_libraries(sil_host rclcpp::rclcpp ${path_smoother_LIBRARIES} ${local_planner_LIBRARIES} ${speed_path_planner_LIBRARIES} ${trajectory_follower_LIBRARIES} ${traffic_signal_stop_LIBRARIES})
target_include_directories(sil_host PRIVATE ${path_smoother_INCLUDE_DIRS} ${local_planner_INCLUDE_DIRS} ${speed_path_planner_INCLUDE_DIRS} ${trajectory_follower_INCLUDE_DIRS} ${traffic_signal_stop_INCLUDE_DIRS})
